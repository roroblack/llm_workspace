def search_product(db: Session, keyword: str) -> dict[str, Any]:
    """상품명 키워드로 상품 목록을 검색한다. 인자: keyword(부분 일치)."""
    rows = db.query(Product).filter(Product.name.contains(keyword)).all()
    results = [{"code": p.product_code, "name": p.name, "price": p.price} for p in rows]
    return {"ok": True, "count": len(results), "results": results}


def get_exchange_rate(db: Session, currency: str) -> dict[str, Any]:
    """통화 코드의 원화 환율을 조회한다. 인자: currency('USD'/'EUR'/'JPY')."""
    rate = EXCHANGE_RATES.get(currency.upper())
    if rate is None:
        return _fail("currency_not_supported", f"지원하지 않는 통화입니다: {currency}")
    return {"ok": True, "currency": currency.upper(), "rate": rate}

def search_product(db: Session, keyword: str) -> dict[str, Any]:
    """상품명 키워드로 상품 목록을 검색한다. 인자: keyword(부분 일치)."""
    rows = db.query(Product).filter(Product.name.contains(keyword)).all()
    results = [{"code": p.product_code, "name": p.name, "price": p.price} for p in rows]
    return {"ok": True, "count": len(results), "results": results}


def get_exchange_rate(db: Session, currency: str) -> dict[str, Any]:
    """통화 코드의 원화 환율을 조회한다. 인자: currency('USD'/'EUR'/'JPY')."""
    rate = EXCHANGE_RATES.get(currency.upper())
    if rate is None:
        return _fail("currency_not_supported", f"지원하지 않는 통화입니다: {currency}")
    return {"ok": True, "currency": currency.upper(), "rate": rate}

- `tests/test_mcp_stdio.py:34-44`: 실제 subprocess에서 서버 예외, 알 수 없는 도구, 잘못된 인자, DB 미준비 시 `is_error=True`가 되는지를 검증하지 않는다. 최소한 한 건씩 추가해 stdio 오류 계약과 폴백 부재를 검증하라.
- `app/mcp/server.py:87-92`, `app/tools/commerce_tools.py:73`: 런타임상 `db`를 사용하지 않아 `None` 전달은 안전하지만 함수 타입이 `Session`이라 정적 계약을 위반한다. 기존 함수의 인자를 `Session | None`으로 바꾸거나, 더 적절하게 미사용 `db` 인자를 제거하고 기존 호출부를 함께 정리하라.

## 낮음

- `tests/test_mcp.py:87-89`: 비밀 검사에서 문자열 `"api_key"` 부재만 확인하므로 실제 키 값이 다른 필드명으로 노출돼도 통과한다. 설정된 sentinel 키 값들이 직렬화 결과에 포함되지 않는지 직접 검사하라.

`with_db`의 `finally` 기반 종료, `rag_qa`/`rag_qa_mod` 이름 분리, subprocess의 `cwd`·`PYTHONPATH`·동일 인터프리터 설정, FastMCP 리소스/프롬프트 데코레이터 시그니처에서는 유효한 결함을 찾지 못했다. 테스트 실행은 코드 문제가 아니라 읽기 전용 환경에서 임시 디렉터리를 만들 수 없어 완료하지 못했다.
